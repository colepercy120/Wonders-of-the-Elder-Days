@holding_illustration_underground_elven = "gfx/interface/illustrations/holding_types/underground/delving.dds"

# ============================================================
# ELVEN UNDERGROUND HALL HOLDINGS
# ============================================================
# Underground counterpart to elven_settlement_01 through elven_settlement_07.
# Gameplay values intentionally match the corresponding elven settlement tier.
# Underground restriction uses the existing LOTR subterranean terrain requirement.
# Presentation temporarily reuses the existing underground holding meshes/illustration.

elven_hall_01 = {
	construction_time = extremely_slow_construction_time

	type_icon = "icon_building_range.dds"

	# ============================================================
	# UNDERGROUND ELVEN HALL MODEL
	# ============================================================
	asset = {
		type = pdxmesh
		name = "dwarven_holding_01_mesh"
		illustration = @holding_illustration_underground_elven
		soundeffect = { soundeffect = "event:/SFX/Ambience/3DMapEmitters/Holdings/Castle/western_castle" soundparameter = { "Tier" = 0 } }
		graphical_cultures = { elven_building_gfx noldor_building_gfx }
		graphical_regions = { graphical_underground }
	}
	is_enabled = {
		OR = {
			building_requirement_elven = yes
			scope:holder = { is_valinor_maiar_character = yes }
		}
		NOT = { # LotR: Preserve the normal elven colony restriction
			county = {
				any_county_province = { has_holding_type = settlement_holding }
			}
		}
	}

	can_construct_potential = {
		lotr_dwarven_underground_buildings_requirement_terrain = yes
		building_requirement_tribal = no
		barony_cannot_construct_holding = no
		custom_tooltip = {
			text = is_not_wastelands_trigger_tooltip
			is_wastelands = no
		}
	}

	can_construct = {
		scope:holder = { is_elf = yes }
		OR = {
			scope:holder = {
				has_medium_hope_or_higher = yes
			}
			scope:holder.top_liege = {
				has_medium_hope_or_higher = yes
			}
			scope:holder.top_liege = {
				is_elf = no
			}
		}
	}

	can_construct_showing_failures_only = {
		building_requirement_tribal_holding_in_county = yes
		county_is_elven = yes
	}

	on_complete = {
		county.holder = { add_elven_hope_effect = { VALUE = major_elven_hope_gain HOPE_EFFECT = effect_elven_hope_elven_refuge_built } }
	}

	cost_gold = extremely_expensive_building_tier_3_cost
	next_building = elven_hall_02

	levy = elven_building_levy_tier_1
	max_garrison = elven_building_max_garrison_tier_1
	garrison_reinforcement_factor = building_garrison_reinforcement_factor_tier_2
	province_modifier = {
		monthly_income = 0.5
		fort_level = great_building_fort_level_tier_1
		hostile_raid_time = 1
		building_slot_add = -1
	}

	province_terrain_modifier = {
		parameter = coastal_holding_bonuses
		is_coastal = yes
		build_speed = -0.1
	}

	province_terrain_modifier = {
		parameter = maritime_mercantilism_coastal_holdings
		is_coastal = yes
		tax_mult = maritime_mercantilism_coastal_holdings_value
	}

	province_terrain_modifier = {
		parameter = coastal_holdings_give_defensive_bonus
		is_coastal = yes
		defender_holding_advantage = 2
	}

	province_terrain_modifier = {
		parameter = elven_holding_fort_level_in_jungle_bonus
		terrain = jungle
		additional_fort_level = castle_holding_fort_level_in_jungle_bonus_value
	}

	province_terrain_modifier = {
		parameter = elven_holding_fort_level_in_deep_forest_bonus
		terrain = deep_forest
		additional_fort_level = castle_holding_fort_level_in_jungle_bonus_value
	}

	province_terrain_modifier = {
		parameter = elven_holding_fort_level_in_mallorn_forest_bonus
		terrain = mallorn_forest
		additional_fort_level = castle_holding_fort_level_in_jungle_bonus_value
	}

	county_modifier = {
		monthly_county_control_growth_add = 0.1
	}

	flag = elven_hall

	ai_value = {
		base = 100
		modifier = {
			factor = 0
			scope:holder = {
				NOT = { is_elf = yes }
			}
		}
	}
}
