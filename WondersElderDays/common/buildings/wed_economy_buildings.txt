artificers_01 = {
	construction_time = slow_construction_time

	can_construct_potential = {
		building_requirement_castle_city_church = { LEVEL = 01 }
		lotr_normal_building_trigger = yes #LotR
		culture = { # Standard unlock from inovation (High medieval)
				has_innovation = innovation_mage_crafting
			}
	}

	can_construct_showing_failures_only = {
		building_requirement_tribal = no
	}
	
	is_enabled = {
		scope:holder = { lotr_normal_building_trigger = yes } #LotR
		is_county_capital = yes
			
	}
	show_disabled = yes
	
	can_construct = {
		
		trigger_if = {
			limit = {
				OR = {
					has_holding_type = castle_holding
					has_holding_type = church_holding
					has_holding_type = temple_citadel_holding
				}
			}
			is_county_capital = yes
		}
		trigger_else = {
			always = no
		}
		OR = {
			culture = { # Standard unlock from inovation (High medieval)
				has_innovation = innovation_mage_crafting
			}
			
		}
	}

	cost_gold = expensive_building_tier_1_cost
	
	province_modifier = {
		monthly_income = excellent_building_tax_tier_1
		tax_mult = good_building_tax_mult_tier_1
		supply_limit_mult = 0.1
	}
	
	county_modifier = {
		development_growth_factor = good_building_development_growth_factor_tier_1
		development_growth = good_building_development_growth_tier_1
	}

	province_government_modifier = { # LotR
		parameter = lotr_government_uses_barter
		monthly_barter_goods = 0.15
	}

	on_complete = {
		#Mandala Creator Aspect
		creator_mandala_built_regular_building_piety_effect = yes
	}
	
	next_building = artificers_02

	type_icon = "icon_building_artificer.dds"
	
	ai_value = {
		base = 20
		ai_tier_1_building_modifier = yes
		ai_tier_1_economical_building_preference_modifier = yes
		ai_general_building_modifier = yes
		directive_to_build_economy_modifier = yes

		modifier = {
			factor = 10
			has_building_or_higher = watermills_01
		}
		modifier = {
			factor = 2
			is_coastal = yes
		}
		modifier = {
			factor = 2
			AND = {
				building_watermills_requirement_terrain = yes
				building_artificers_requirement_terrain = yes
			}
		}
	}
}
