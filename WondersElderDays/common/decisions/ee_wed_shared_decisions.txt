decision_reclaim_gondolin = {
	title = decision_reclaim_gondolin
	desc = decision_reclaim_gondolin_desc
	confirm_text = decision_reclaim_gondolin_confirm
	selection_tooltip = decision_reclaim_gondolin_tooltip
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_elves_01.dds"
	}
	

	# AI
	ai_goal = yes
	ai_potential = {
		always = yes
	}
	ai_will_do = {
		base = 100
	}
	
	# IS SHOWN
	decision_group_type = realms_in_exile_major
	
	is_shown = {
		is_elf = yes
		OR = {
			
			
				
			completely_controls = title:d_andunaloni # Completely controls the old heartland of the Kingdom
			
			
			this = character:lineoffeanor2
			this = character:lineofglorfindel
		}
		
		NOT = {
			is_target_in_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:decision_reclaim_gondolin
			}		
		}
	}
	
	# IS VALID
	is_valid = {
		prestige_level >= 5
		piety_level >= 5
		completely_controls = title:d_andunaloni # Completely controls the old heartland of the Kingdom
		
		has_high_hope_or_higher = yes
		
		province:4335 = {
			has_building_or_higher = wonder_gondolin_02
		}
	}

	is_valid_showing_failures_only = {
		is_imprisoned = no
		is_available_adult = yes
		is_at_war = no
	}

	# EFFECT
	effect = {
		add_prestige = 1000
		root.dynasty = { add_dynasty_prestige = 1000 }
		add_elven_hope_global_effect = { VALUE = 100 HOPE_EFFECT = effect_elven_hope_good_realm_rises }
		
		title:d_andunaloni = {
			set_de_jure_liege_title = title:k_gondolin
			every_de_jure_county = {
				change_development_level = 5
			}
			set_capital_county = title:c_anach
		}
		
		
		if = {
			limit = { highest_held_title_tier < tier_empire }
			set_primary_title_to = title:k_gondolin
		}
		title:k_gondolin = {
			set_de_jure_liege_title = title:e_lindon
		}
		title:k_gondolin = {
			set_capital_county = title:c_anach
			
		}
		#Create the title and make it primary
		create_title_and_vassal_change = {
			type = created
			save_scope_as = title_change
			add_claim_on_loss = yes
		}
		title:k_gondolin = {
			change_title_holder = {
				holder = root
				change = scope:title_change
			}
		}
		resolve_title_and_vassal_change = scope:title_change
		
		add_to_global_variable_list = {
			name = unavailable_unique_decisions
			target = flag:decision_reclaim_gondolin
		}
		
		

		trigger_event = {
			id = ee_wed_shared.01	# Reclaim Nenuial Kingdom
		}
	}
}
