on_game_start = {
	on_actions = {
		correct_alignment_modifier
	}
}

yearly_global_pulse = {
	on_actions = {
		percival_yearly_pulse
	}
}

percival_yearly_pulse = {
	effect = {
		trigger_event = { on_action = correct_alignment_modifier }
		trigger_event = { on_action = correct_alignment_modifier months = 1 }
		trigger_event = { on_action = correct_alignment_modifier months = 2 }
		trigger_event = { on_action = correct_alignment_modifier months = 3 }
		trigger_event = { on_action = correct_alignment_modifier months = 4 }
		trigger_event = { on_action = correct_alignment_modifier months = 5 }
		trigger_event = { on_action = correct_alignment_modifier months = 6 }
		trigger_event = { on_action = correct_alignment_modifier months = 7 }
		trigger_event = { on_action = correct_alignment_modifier months = 8 }
		trigger_event = { on_action = correct_alignment_modifier months = 9 }
		trigger_event = { on_action = correct_alignment_modifier months = 10 }
		trigger_event = { on_action = correct_alignment_modifier months = 11 }
	}
}
correct_alignment_modifier = {
	effect = {
		every_living_character = {
			limit = {
				has_character_modifier = good_racial_hatred
				faith_is_good = yes
			}
			remove_character_modifier = good_racial_hatred
		}
		every_living_character = {
			limit = {
				has_character_modifier = evil_racial_hatred
				faith_is_evil = yes
			}
			remove_character_modifier = evil_racial_hatred
		}
	}
}
